hs.hotkey.bind({"alt", "shift"}, "X", function()
    local activeID = hs.spaces.focusedSpace()
    local screen = hs.screen.mainScreen()
    local screenUUID = screen:getUUID()
    local screenIDs = hs.spaces.allSpaces()[screenUUID]

    if #screenIDs < 2 then
        return
    end
    
    hs.spaces.gotoSpace(screenIDs[1])
    
    hs.timer.usleep(300000)
    
    local center = hs.geometry.rectMidPoint(screen:frame())
    hs.mouse.setAbsolutePosition(center)
    local newSpace = hs.spaces.focusedSpace()
    
    if newSpace ~= activeID then
        hs.spaces.removeSpace(activeID)
        screenIDs = hs.spaces.allSpaces()[screenUUID]
        hs.spaces.gotoSpace(screenIDs[#screenIDs])
    else
        hs.alert("Failed to switch spaces - cannot remove")
    end
end)

hs.hotkey.bind({"cmd", "alt", "ctrl"}, "C", function()
    -- 1. Create 3 new Arc windows via AppleScript
    local script = [[
        tell application "Arc"
            repeat 3 times
                make new window
            end repeat
            activate
        end tell
    ]]
    hs.osascript.applescript(script)

    -- 2. Give Arc a moment to spawn windows
    hs.timer.doAfter(1.5, function()
        local app = hs.appfinder.appFromName("Arc")
        if not app then return end

        local windows = app:allWindows()
        local screens = hs.screen.allScreens()

        -- 3. Sort windows by creation time (rough order)
        table.sort(windows, function(a, b)
            return a:id() < b:id()
        end)

        -- 4. Move each window to a different screen
        for i, win in ipairs(windows) do
            local targetScreen = screens[((i - 1) % #screens) + 1]
            if win:isStandard() then
                win:moveToScreen(targetScreen)
                win:centerOnScreen(targetScreen)
            end
        end
    end)
end)

local super = {"ctrl", "alt", "cmd"}

local function orderedScreens()
  local laptop
  local others = {}
  for _, s in ipairs(hs.screen.allScreens()) do
    if not laptop and s:name():match("Built%-in") then
      laptop = s
    else
      table.insert(others, s)
    end
  end
  table.sort(others, function(a, b) return a:frame().x < b:frame().x end)
  if laptop then table.insert(others, 1, laptop) end
  return others
end

local function moveToScreen(win, target, duration)
    local unit = win:frame():toUnitRect(win:screen():frame())
    local newFrame = unit:fromUnitRect(target:frame())

    -- Chromium/Electron apps break animated resizes while AXEnhancedUserInterface is on
    local axApp = hs.axuielement.applicationElement(win:application())
    local wasEnhanced = axApp and axApp.AXEnhancedUserInterface
    if wasEnhanced then axApp.AXEnhancedUserInterface = false end

    win:setFrameInScreenBounds(newFrame, duration)

    if wasEnhanced then
      hs.timer.doAfter((duration or hs.window.animationDuration) + 0.1, function()
        axApp.AXEnhancedUserInterface = true
      end)
    end
end

-- held at module level so pending timers aren't garbage-collected before firing
local fullScreenMoveTimer

local function sendTo(i)
  return function()
    local win = hs.window.focusedWindow()
    local target = orderedScreens()[i]
    if not (win and target) or win:screen():id() == target:id() then return end

    if not win:isFullScreen() then return moveToScreen(win, target) end

    win:setFullScreen(false)
    local polls = 0
    -- moves sent during the exit animation are often dropped, so keep retrying until it lands
    fullScreenMoveTimer = hs.timer.doEvery(0.05, function()
      polls = polls + 1
      if polls > 80 then return fullScreenMoveTimer:stop() end
      if win:isFullScreen() then return end
      if win:screen():id() ~= target:id() then return moveToScreen(win, target, 0) end
      fullScreenMoveTimer:stop()
      win:setFullScreen(true)
    end)
  end
end

hs.hotkey.bind(super, "[",  sendTo(1)) -- laptop
hs.hotkey.bind(super, "]",  sendTo(2)) -- middle
hs.hotkey.bind(super, "\\", sendTo(3)) -- last

local CORNER_INSET = 40

local function clickConductorCorner()
  local app = hs.application.get("com.conductor.app")
  local win = app and (app:mainWindow() or app:allWindows()[1])
  if not win or win:isMinimized() then return end

  local original = hs.mouse.absolutePosition()
  local f = win:screen():frame()
  hs.eventtap.leftClick(hs.geometry.point(f.x + f.w - CORNER_INSET, f.y + f.h - CORNER_INSET))
  hs.mouse.absolutePosition(original)
end

hs.hotkey.bind(super, "T", clickConductorCorner)
hs.hotkey.bind(super, "right", clickConductorCorner)
